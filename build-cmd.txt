apk:
fvm flutter build apk --release \
  --dart-define=ENV=prod \
  --dart-define=ENABLE_FIREBASE=true \
  --dart-define=ENABLE_AUTH=true \
  --dart-define=ENABLE_CART=true \
  --dart-define=ENABLE_PUSH_NOTIFICATIONS=true \
  --dart-define=ENABLE_CHAT=true \
  --dart-define=ENABLE_LOCALIZATION=true \
  --dart-define=ENABLE_WORKMANAGER=true \
  --dart-define=ENABLE_GO_ROUTER=true

default run:
fvm flutter run --flavor template \
  --dart-define=APP_FLAVOR=template \
  --dart-define=BRAND_ID=template

default build apk:
fvm flutter build apk --flavor template --release \
  --no-tree-shake-icons \
  --dart-define=APP_FLAVOR=template \
  --dart-define=BRAND_ID=template

default build ios:
fvm flutter build ios --flavor template --release --no-codesign \
  --no-tree-shake-icons \
  --dart-define=APP_FLAVOR=template \
  --dart-define=BRAND_ID=template
